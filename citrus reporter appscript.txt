// ── CORE PROCESSING ──────────────────────────────────────────
function processFolderIntoSheet(folderId, codes, ss, reportTabName) {
  var folder = DriveApp.getFolderById(folderId);
  var latestFiles = {};
  var files = folder.getFilesByType(MimeType.PDF);
  while (files.hasNext()) {
    var file = files.next();
    var name = file.getName().toUpperCase();
    for (var i = 0; i < codes.length; i++) {
      if (name.indexOf(codes[i]) !== -1) {
        var code = codes[i];
        var modified = file.getLastUpdated().getTime();
        if (!latestFiles[code] || modified > latestFiles[code].time) {
          latestFiles[code] = { file: file, time: modified, name: file.getName() };
        }
        break;
      }
    }
  }

  var props = PropertiesService.getScriptProperties();
  var anyNew = false;
  for (var code in latestFiles) {
    if (latestFiles[code].time > parseInt(props.getProperty("last_" + code) || "0")) {
      anyNew = true; break;
    }
  }
  if (!anyNew) { Logger.log("No new PDFs for " + reportTabName); return; }

  var allData = {};
  for (var code in latestFiles) {
    var entry = latestFiles[code];
    Logger.log("Processing " + code + ": " + entry.name);

    var data = extractViaService(entry.file, code);
    if (!data) { Logger.log("Failed to extract: " + entry.name); continue; }

    Logger.log(code + ": " + data.lines.length + " price lines, sold=" + data.totalSoldToday + ", yesterday=" + data.yesterdayAvailable + ", received=" + data.receivedToday + ", value=" + data.totalValueSold);

    writeToTab(ss, code, data);
    allData[code] = data;
    props.setProperty("last_" + code, String(entry.time));

    // Move processed file to processed folder
    try {
      var processedFolder = DriveApp.getFolderById(CONFIG.PROCESSED_FOLDER_ID);
      entry.file.moveTo(processedFolder);
      Logger.log("Moved " + entry.name + " to processed folder.");
    } catch(e) {
      Logger.log("Could not move " + entry.name + ": " + e.message);
    }
  }

  if (Object.keys(allData).length === 0) return;
  writeReportTab(ss, reportTabName, buildWhatsAppMessage(allData, codes));
  Logger.log(reportTabName + " updated.");
}

// ── EXTRACT VIA PYTHON SERVICE ───────────────────────────────
function extractViaService(file, commodityCode) {
  try {
    var blob = file.getBlob();
    var base64 = Utilities.base64Encode(blob.getBytes());

    var response = UrlFetchApp.fetch(CONFIG.EXTRACTOR_URL, {
      method: "POST",
      contentType: "application/json",
      payload: JSON.stringify({
        pdf_base64: base64,
        commodity_code: commodityCode,
      }),
      muteHttpExceptions: true,
    });

    var code = response.getResponseCode();
    if (code !== 200) {
      Logger.log("Service error (" + code + "): " + response.getContentText());
      return null;
    }

    var data = JSON.parse(response.getContentText());

    // Calculate bag/box split for ORGS
    if (commodityCode === "ORGS" && data.lines) {
      // Note: bag/box split needs per-row qty data which isn't in the simplified
      // service response yet. Fallback: calculate from totalSoldToday/yesterdayAvailable
      // proportionally isn't accurate, so we skip this for now unless service is extended.
      data.bagPct = null;
      data.boxPct = null;
    }

    return data;
  } catch (e) {
    Logger.log("extractViaService error: " + e.message);
    return null;
  }
}

// ── HELPERS ──────────────────────────────────────────────────
function formatNum(n) {
  if (!n) return "0";
  return Math.round(n).toLocaleString();
}

function formatRand(n) {
  if (!n) return "R0.00";
  return "R" + parseFloat(n).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ",");
}

function toTitleCase(str) {
  if (!str) return "";
  return str.toLowerCase().replace(/\b\w/g, function(c) { return c.toUpperCase(); });
}

function toPackLabel(key) {
  key = key.replace(/ EMDASH /g, " - ");
  key = key.replace(/(\d+\.?\d*)(KG)/gi, function(_, n) { return n + "kg"; });
  key = key.replace(/\b([A-Z]{2,})\b/g, function(w) { return toTitleCase(w); });
  return key;
}

// ── WRITE TO SHEET ────────────────────────────────────────────
function writeToTab(ss, code, data) {
  var tab = ss.getSheetByName(code) || ss.insertSheet(code);
  tab.clearContents();
  tab.getRange(1,1).setValue(code + " - " + data.date).setFontWeight("bold").setFontSize(12);
  tab.getRange(2,1,1,4).setValues([["Sold Today","Yesterday Available","Received Today","Total Value"]]).setFontWeight("bold").setBackground("#f0f0f0");
  tab.getRange(3,1,1,4).setValues([[data.totalSoldToday, data.yesterdayAvailable, data.receivedToday, data.totalValueSold]]);
  tab.getRange(5,1,1,4).setValues([["Pack Type","Variety","Class/Grade","Avg Price (R)"]]).setFontWeight("bold").setBackground("#fff0e0");
  if (data.lines.length > 0) {
    tab.getRange(6,1,data.lines.length,4).setValues(
      data.lines.map(function(l) { return [l.packType, l.variety, l.classGrade, l.avgPrice]; })
    );
  }
  tab.autoResizeColumns(1, 4);
}

function writeReportTab(ss, tabName, message) {
  var tab = ss.getSheetByName(tabName) || ss.insertSheet(tabName);
  tab.clearContents();
  tab.getRange(1,1).setValue("Generated: " + new Date().toLocaleString()).setFontWeight("bold");
  tab.getRange(2,1).setValue(message).setWrap(true).setFontFamily("Courier New").setFontSize(10);
  tab.setColumnWidth(1, 500);
  tab.setRowHeight(2, Math.min(2000, message.split("\n").length * 18 + 20));
}

// ── WHATSAPP MESSAGE ──────────────────────────────────────────
function buildWhatsAppMessage(allData, codes) {
  var lines = [], date = "";
  var grandSold = 0, grandYesterday = 0, grandReceived = 0, grandValue = 0;
  var presentCodes = [];

  for (var i = 0; i < codes.length; i++) {
    if (allData[codes[i]]) { date = allData[codes[i]].date; break; }
  }

  lines.push("EMOJI_ORANGE JHB FRESH PRODUCE MARKET");
  lines.push("EMOJI_CAL " + date);
  lines.push("");

  // ORGS bag/box split line right after header (if available)
  for (var i = 0; i < codes.length; i++) {
    var d0 = allData[codes[i]];
    if (d0 && codes[i] === "ORGS" && d0.bagPct !== null && d0.bagPct !== undefined) {
      lines.push("ORGS - Bags: " + d0.bagPct + "% | Boxes: " + d0.boxPct + "%");
      lines.push("");
      break;
    }
  }

  for (var i = 0; i < codes.length; i++) {
    var code = codes[i];
    var d = allData[code];
    if (!d) continue;
    presentCodes.push(code);
    grandSold      += d.totalSoldToday || 0;
    grandYesterday += d.yesterdayAvailable || 0;
    grandReceived  += d.receivedToday || 0;
    grandValue     += d.totalValueSold || 0;

    lines.push("EMOJI_ORANGE " + (COMMODITY_NAMES[code] || code));
    lines.push("EMOJI_BOX Sold: " + formatNum(d.totalSoldToday) + " | Yesterday: " + formatNum(d.yesterdayAvailable) + " | Received Today: " + formatNum(d.receivedToday));

    var groups = {}, groupOrder = [];
    for (var j = 0; j < d.lines.length; j++) {
      var l = d.lines[j];
      var key = l.packType + (l.variety ? " EMDASH " + toTitleCase(l.variety) : "");
      if (!groups[key]) { groups[key] = []; groupOrder.push(key); }
      groups[key].push(l);
    }
    for (var k = 0; k < groupOrder.length; k++) {
      lines.push(toPackLabel(groupOrder[k]));
      var items = groups[groupOrder[k]];
      for (var m = 0; m < items.length; m++) {
        lines.push("BULLET " + items[m].classGrade + " EMDASH " + formatRand(items[m].avgPrice));
      }
      lines.push("");
    }
  }

  lines.push("EMOJI_CHART GRAND TOTAL (" + presentCodes.join(" / ") + ")");
  lines.push("EMOJI_MONEY Total Value Sold EMDASH " + formatRand(grandValue));
  lines.push("EMOJI_BOX Total Sold Today EMDASH " + formatNum(grandSold));
  lines.push("EMOJI_BOX Total Yesterday Available EMDASH " + formatNum(grandYesterday));
  lines.push("EMOJI_BOX Total Received Today EMDASH " + formatNum(grandReceived));
  lines.push("EMOJI_PIN Johannesburg Fresh Produce Market");
  lines.push("EMOJI_CAL " + date);

  return lines.join("\n");
}

function restoreEmojis(text) {
  if (!text) return text;
  var s = String(text);
  s = s.replace(/EMOJI_ORANGE /g, "\uD83C\uDF4A ");
  s = s.replace(/EMOJI_CAL /g,    "\uD83D\uDCC5 ");
  s = s.replace(/EMOJI_BOX /g,    "\uD83D\uDCE6 ");
  s = s.replace(/EMOJI_CHART /g,  "\uD83D\uDCCA ");
  s = s.replace(/EMOJI_MONEY /g,  "\uD83D\uDCB0 ");
  s = s.replace(/EMOJI_PIN /g,    "\uD83D\uDCCD ");
  s = s.replace(/BULLET /g,       "\u2022 ");
  s = s.replace(/ EMDASH /g,      " \u2014 ");
  return s;
}

// ── WEB APP ──────────────────────────────────────────────────
function doGet() {
  var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
  var softMsg = getReportMessage(ss, "SOFT_REPORT");
  var hardMsg = getReportMessage(ss, "HARD_REPORT");
  var html = buildWebApp(softMsg, hardMsg, new Date().toLocaleString("en-ZA"));
  return HtmlService.createHtmlOutput(html)
    .setTitle("Citrus Reporter")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function getReportMessage(ss, tabName) {
  var tab = ss.getSheetByName(tabName);
  if (!tab) return "No report yet. Drop PDFs in the folder and wait.";
  var val = tab.getRange(2,1).getValue();
  if (!val) return "No report yet. Drop PDFs in the folder and wait.";
  return restoreEmojis(String(val));
}
